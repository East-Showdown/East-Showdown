RUS_mastery_init = {
	if = {
		limit = {
			NOT = { has_country_flag = RUS_mastery_program_active }
		}
		set_country_flag = RUS_mastery_program_active

		set_variable = { RUS_mastery_btr_lead = 1 }
		set_variable = { RUS_mastery_btr_stage = 3 }
		set_variable = { RUS_mastery_btr_days = 0 }
		set_variable = { RUS_mastery_bmp_lead = 3 }

		set_variable = { RUS_mastery_bmd_stage = 3 }
		set_variable = { RUS_mastery_bmd_days = 0 }

		add_ideas = RUS_mastery_btr80_3
		add_ideas = RUS_mastery_btr70_1
		add_ideas = RUS_mastery_btr87_1
		add_ideas = RUS_mastery_btr22_1
		add_ideas = RUS_mastery_btr90_1
		add_ideas = RUS_mastery_btr16_1
		add_ideas = RUS_mastery_bmp1_1
		add_ideas = RUS_mastery_bmp2_1
		add_ideas = RUS_mastery_bmp3_3
		add_ideas = RUS_mastery_bmpk25_1
		add_ideas = RUS_mastery_bmpt15_1
		add_ideas = RUS_mastery_bmd2_1
		add_ideas = RUS_mastery_bmd3_1
		add_ideas = RUS_mastery_bmd4_3
	}
}

RUS_mastery_daily_tick = {
	if = {
		limit = {
			original_tag = RUS
			has_country_flag = RUS_mastery_program_active
		}
		if = {
			limit = {
				check_variable = { RUS_mastery_btr_lead > 0 }
				check_variable = { RUS_mastery_btr_stage < 3 }
			}
			add_to_variable = { RUS_mastery_btr_days = 1 }
			if = {
				limit = {
					check_variable = { RUS_mastery_btr_stage = 1 }
					check_variable = {
						var = RUS_mastery_btr_days
						value = 90
						compare = greater_than_or_equals
					}
				}
				set_variable = { RUS_mastery_btr_stage = 2 }
				RUS_mastery_btr_advance = yes
			}
			else_if = {
				limit = {
					check_variable = { RUS_mastery_btr_stage = 2 }
					check_variable = {
						var = RUS_mastery_btr_days
						value = 180
						compare = greater_than_or_equals
					}
				}
				set_variable = { RUS_mastery_btr_stage = 3 }
				RUS_mastery_btr_advance = yes
			}
		}
		if = {
			limit = {
				check_variable = { RUS_mastery_bmp_lead > 0 }

			}

		if = {
