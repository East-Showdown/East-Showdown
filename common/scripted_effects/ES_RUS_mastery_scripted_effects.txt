RUS_mastery_init = {

		limit = {
			NOT = { has_country_flag = RUS_mastery_program_active }

# Этапы повышает само решение RUS_mastery_*_progress по истечении days_remove.
RUS_mastery_daily_tick = {
	RUS_artillery_program_reconcile = yes
	RUS_spa_migrate_direct_bonuses = yes
	if = {
		limit = {
			original_tag = RUS
			NOT = { has_country_flag = RUS_hidden_ideas_reregistered_v2 }
		}
		RUS_program_ideas_reregister = yes
		set_country_flag = RUS_hidden_ideas_reregistered_v2
	}
	if = {
		limit = {
			original_tag = RUS
			has_country_flag = RUS_mastery_program_active
		}

		# Разовая миграция старых сохранений: штрафы за образцы вынесены в отдельные идеи
		if = {
			limit = { NOT = { has_country_flag = RUS_samples_split_migrated } }
			add_ideas = {
				RUS_samples_of_equipment_kurganets25
				RUS_samples_of_equipment_btr16
				RUS_samples_of_equipment_t15
			}
			set_country_flag = RUS_samples_split_migrated
		}

		if = {
			limit = {
				check_variable = { RUS_mastery_btr_lead > 0 }
				check_variable = { RUS_mastery_btr_stage < 3 }
			}
			if = {
				limit = { NOT = { has_decision = RUS_mastery_btr_progress } }
				RUS_mastery_btr_timer_restart = yes
			}
			else = { add_to_variable = { RUS_mastery_btr_days = 1 } }
			if = { limit = { check_variable = { RUS_mastery_btr_stage = 1 } } set_variable = { RUS_mastery_btr_remaining = 90 } }
			else = { set_variable = { RUS_mastery_btr_remaining = 180 } }
			subtract_from_variable = { RUS_mastery_btr_remaining = RUS_mastery_btr_days }
			clamp_variable = { var = RUS_mastery_btr_remaining min = 0 }
		}

		if = {
			limit = {
				check_variable = { RUS_mastery_bmp_lead > 0 }
				check_variable = { RUS_mastery_bmp_stage < 3 }
			}
			if = {
				limit = { NOT = { has_decision = RUS_mastery_bmp_progress } }
				RUS_mastery_bmp_timer_restart = yes
			}
			else = { add_to_variable = { RUS_mastery_bmp_days = 1 } }
			if = { limit = { check_variable = { RUS_mastery_bmp_stage = 1 } } set_variable = { RUS_mastery_bmp_remaining = 90 } }
			else = { set_variable = { RUS_mastery_bmp_remaining = 180 } }
			subtract_from_variable = { RUS_mastery_bmp_remaining = RUS_mastery_bmp_days }
			clamp_variable = { var = RUS_mastery_bmp_remaining min = 0 }
		}

		if = {
			limit = {
				check_variable = { RUS_mastery_bmd_lead > 0 }
				check_variable = { RUS_mastery_bmd_stage < 3 }
			}
			if = {
				limit = { NOT = { has_decision = RUS_mastery_bmd_progress } }
				RUS_mastery_bmd_timer_restart = yes
			}
			else = { add_to_variable = { RUS_mastery_bmd_days = 1 } }
			if = { limit = { check_variable = { RUS_mastery_bmd_stage = 1 } } set_variable = { RUS_mastery_bmd_remaining = 90 } }
			else = { set_variable = { RUS_mastery_bmd_remaining = 180 } }
			subtract_from_variable = { RUS_mastery_bmd_remaining = RUS_mastery_bmd_days }
			clamp_variable = { var = RUS_mastery_bmd_remaining min = 0 }
		}
	}

# Remove the inactive artillery programme left by the old direction-switch effect.
RUS_artillery_program_reconcile = {
	if = {
		limit = { original_tag = RUS }
		if = {
			limit = { NOT = { has_country_flag = RUS_artillery_program_state_v2 } }
			set_country_flag = RUS_artillery_program_state_v2
			set_variable = { RUS_artillery_program_direction = 0 }
			set_variable = { RUS_artillery_program_stage = 0 }
			if = {
				limit = {
					has_country_flag = RUS_artillery_mobile_choice
				}
				set_variable = { RUS_artillery_program_direction = 1 }
			}
			else_if = {
				limit = {
					has_country_flag = RUS_artillery_heavy_choice
				}
				set_variable = { RUS_artillery_program_direction = 2 }
			}
			else_if = {
				limit = {
					OR = {
						has_idea = RUS_artillery_mobile_program
						has_idea = RUS_artillery_mobile_program_15
						has_idea = RUS_artillery_mobile_program_20
					}
				}
				set_variable = { RUS_artillery_program_direction = 1 }
			}
			else_if = {
				limit = {
					OR = {
						has_idea = RUS_artillery_heavy_program
						has_idea = RUS_artillery_heavy_program_15
						has_idea = RUS_artillery_heavy_program_20
					}
				}
				set_variable = { RUS_artillery_program_direction = 2 }
			}
			if = {
				limit = { check_variable = { RUS_artillery_program_direction > 0 } }
				set_variable = { RUS_artillery_program_stage = 1 }
				if = {
					limit = {
						OR = {
							AND = {
								check_variable = { RUS_artillery_program_direction = 1 }
								has_idea = RUS_artillery_mobile_program_20
							}
							AND = {
								check_variable = { RUS_artillery_program_direction = 2 }
								has_idea = RUS_artillery_heavy_program_20
							}
						}
					}
					set_variable = { RUS_artillery_program_stage = 3 }
				}
				else_if = {
					limit = {
						OR = {
							AND = {
								check_variable = { RUS_artillery_program_direction = 1 }
								has_idea = RUS_artillery_mobile_program_15
							}
							AND = {
								check_variable = { RUS_artillery_program_direction = 2 }
								has_idea = RUS_artillery_heavy_program_15
							}
						}
					}
					set_variable = { RUS_artillery_program_stage = 2 }
				}
				# Broken legacy timers cannot prove elapsed time; start a full safe interval once.
				if = {
					limit = { NOT = { has_country_flag = RUS_artillery_program_last_change } }
					set_country_flag = RUS_artillery_program_last_change
				}
			}
			clr_country_flag = RUS_artillery_direction_cooldown
		}

		if = {
			limit = { NOT = { has_country_flag = RUS_artillery_program_timed_v3 } }
			set_country_flag = RUS_artillery_program_timed_v3
			clr_country_flag = RUS_artillery_program_last_change
			clr_country_flag = RUS_artillery_direction_cooldown
			clr_country_flag = RUS_artillery_program_in_progress
			set_variable = { RUS_artillery_pending_direction = 0 }
			set_variable = { RUS_artillery_pending_stage = 0 }
		}
		# A removed or cancelled native decision must not leave the programme locked.
		if = {
			limit = {
				has_country_flag = RUS_artillery_program_in_progress
				NOT = {
					OR = {
						has_decision = RUS_switch_to_mobile_artillery
						has_decision = RUS_upgrade_mobile_artillery
						has_decision = RUS_upgrade_mobile_artillery_2
						has_decision = RUS_switch_to_heavy_artillery
						has_decision = RUS_upgrade_heavy_artillery
						has_decision = RUS_upgrade_heavy_artillery_2
					}
				}
			}
			clr_country_flag = RUS_artillery_program_in_progress
			set_variable = { RUS_artillery_pending_direction = 0 }
			set_variable = { RUS_artillery_pending_stage = 0 }
		}
		if = {
			limit = { check_variable = { RUS_artillery_program_direction = 1 } }
			if = { limit = { NOT = { has_country_flag = RUS_artillery_mobile_choice } } set_country_flag = RUS_artillery_mobile_choice }
			clr_country_flag = RUS_artillery_heavy_choice
		}
		else_if = {
			limit = { check_variable = { RUS_artillery_program_direction = 2 } }
			if = { limit = { NOT = { has_country_flag = RUS_artillery_heavy_choice } } set_country_flag = RUS_artillery_heavy_choice }
			clr_country_flag = RUS_artillery_mobile_choice
		}
		else = {
			clr_country_flag = RUS_artillery_mobile_choice
			clr_country_flag = RUS_artillery_heavy_choice
		}
		if = {
			limit = {
				has_idea = RUS_artillery_mobile_program
				NOT = {
					AND = {
						check_variable = { RUS_artillery_program_direction = 1 }
						check_variable = { RUS_artillery_program_stage = 1 }
					}
				}
			}
			remove_ideas = RUS_artillery_mobile_program
		}
		if = {
			limit = {
				check_variable = { RUS_artillery_program_direction = 1 }
				check_variable = { RUS_artillery_program_stage = 1 }
				NOT = { has_idea = RUS_artillery_mobile_program }
			}
			add_ideas = RUS_artillery_mobile_program
		}
		if = {
			limit = {
				has_idea = RUS_artillery_mobile_program_15
				NOT = {
					AND = {
						check_variable = { RUS_artillery_program_direction = 1 }
						check_variable = { RUS_artillery_program_stage = 2 }
					}
				}
			}
			remove_ideas = RUS_artillery_mobile_program_15
		}
		if = {
			limit = {
				check_variable = { RUS_artillery_program_direction = 1 }
				check_variable = { RUS_artillery_program_stage = 2 }
				NOT = { has_idea = RUS_artillery_mobile_program_15 }
			}
			add_ideas = RUS_artillery_mobile_program_15
		}
		if = {
			limit = {
				has_idea = RUS_artillery_mobile_program_20
				NOT = {
					AND = {
						check_variable = { RUS_artillery_program_direction = 1 }
						check_variable = { RUS_artillery_program_stage = 3 }
					}
				}
			}
			remove_ideas = RUS_artillery_mobile_program_20
		}
		if = {
			limit = {
				check_variable = { RUS_artillery_program_direction = 1 }
				check_variable = { RUS_artillery_program_stage = 3 }
				NOT = { has_idea = RUS_artillery_mobile_program_20 }
			}
			add_ideas = RUS_artillery_mobile_program_20
		}
		if = {
			limit = {
				has_idea = RUS_artillery_heavy_program
				NOT = {
					AND = {
						check_variable = { RUS_artillery_program_direction = 2 }
						check_variable = { RUS_artillery_program_stage = 1 }
					}
				}
			}
			remove_ideas = RUS_artillery_heavy_program
		}
		if = {
			limit = {
				check_variable = { RUS_artillery_program_direction = 2 }
				check_variable = { RUS_artillery_program_stage = 1 }
				NOT = { has_idea = RUS_artillery_heavy_program }
			}
			add_ideas = RUS_artillery_heavy_program
		}
		if = {
			limit = {
				has_idea = RUS_artillery_heavy_program_15
				NOT = {
					AND = {
						check_variable = { RUS_artillery_program_direction = 2 }
						check_variable = { RUS_artillery_program_stage = 2 }
					}
				}
			}
			remove_ideas = RUS_artillery_heavy_program_15
		}
		if = {
			limit = {
				check_variable = { RUS_artillery_program_direction = 2 }
				check_variable = { RUS_artillery_program_stage = 2 }
				NOT = { has_idea = RUS_artillery_heavy_program_15 }
			}
			add_ideas = RUS_artillery_heavy_program_15
		}
		if = {
			limit = {
				has_idea = RUS_artillery_heavy_program_20
				NOT = {
					AND = {
						check_variable = { RUS_artillery_program_direction = 2 }
						check_variable = { RUS_artillery_program_stage = 3 }
					}
				}
			}
			remove_ideas = RUS_artillery_heavy_program_20
		}
		if = {
			limit = {
				check_variable = { RUS_artillery_program_direction = 2 }
				check_variable = { RUS_artillery_program_stage = 3 }
				NOT = { has_idea = RUS_artillery_heavy_program_20 }
			}
			add_ideas = RUS_artillery_heavy_program_20
		}
	}
}
