# Вызывается в скоупе Украины при начале войны или загрузке военного сценария.
ES_UKR_initialize_eastern_fortified_region = {
	if = {
		limit = {
			original_tag = UKR
			is_ai = yes
			date < 2022.8.23
			NOT = { has_country_flag = ES_UKR_eastern_fortified_region_initialized }
		}
		# Общий крайний срок — 23 августа, 180 дней с 24 февраля 2022.
		set_variable = { ES_UKR_eastern_fortified_region_days = 180 }
		if = {
			limit = { has_start_date = 2022.04.15 }
			set_variable = { ES_UKR_eastern_fortified_region_days = 130 }
		}
		715 = { add_dynamic_modifier = { modifier = ES_UKR_eastern_fortified_region scope = UKR days = UKR.ES_UKR_eastern_fortified_region_days } }
		893 = { add_dynamic_modifier = { modifier = ES_UKR_eastern_fortified_region scope = UKR days = UKR.ES_UKR_eastern_fortified_region_days } }
		890 = { add_dynamic_modifier = { modifier = ES_UKR_eastern_fortified_region scope = UKR days = UKR.ES_UKR_eastern_fortified_region_days } }
		165 = { add_dynamic_modifier = { modifier = ES_UKR_eastern_fortified_region scope = UKR days = UKR.ES_UKR_eastern_fortified_region_days } }
		900 = { add_dynamic_modifier = { modifier = ES_UKR_eastern_fortified_region scope = UKR days = UKR.ES_UKR_eastern_fortified_region_days } }
		981 = { add_dynamic_modifier = { modifier = ES_UKR_eastern_fortified_region scope = UKR days = UKR.ES_UKR_eastern_fortified_region_days } }
		911 = { add_dynamic_modifier = { modifier = ES_UKR_eastern_fortified_region scope = UKR days = UKR.ES_UKR_eastern_fortified_region_days } }
		912 = { add_dynamic_modifier = { modifier = ES_UKR_eastern_fortified_region scope = UKR days = UKR.ES_UKR_eastern_fortified_region_days } }
		set_country_flag = ES_UKR_eastern_fortified_region_initialized
	}
}
