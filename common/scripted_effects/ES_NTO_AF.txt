NTO_try_refugee_help_centers = {
	if = {
		limit = {
			has_completed_focus = nto_help_ukrainian_refugees
			NOT = {
				has_completed_focus = nto_refugee_help_centers
			}
			NOT = {
				has_shine_effect_on_focus = nto_refugee_help_centers
			}
			NOT = {
				has_country_flag = nto_refugee_help_centers_in_progress_flag
			}
		}
		country_event = {
			id = nto_refugee_auto_focus.1
			days = 0
		}
	}
}
