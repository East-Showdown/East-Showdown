UKR_start_steel_general_war_timer = {
	if = {
		limit = { NOT = { has_country_flag = UKR_steel_general_war_timer_started } }
		set_country_flag = UKR_steel_general_war_timer_started
		set_variable = { UKR_steel_general_unlock_day = global.num_days }
		add_to_variable = { UKR_steel_general_unlock_day = 180 }
		UKR_update_steel_general_war_timer = yes
	}
}

UKR_update_steel_general_war_timer = {
	set_variable = { UKR_steel_general_days_remaining = UKR_steel_general_unlock_day }
	subtract_from_variable = { UKR_steel_general_days_remaining = global.num_days }
	clamp_variable = { var = UKR_steel_general_days_remaining min = 0 max = 180 }
}
