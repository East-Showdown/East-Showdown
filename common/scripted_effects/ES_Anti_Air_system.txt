	# =========================================================

	# =========================================================

		limit = {
			check_variable = {
				var = SAM_mobile_groups_count
				value = 0
				compare = greater_than
			}
			check_variable = {
				var = SAM_targets_remaining
				value = 0
				compare = greater_than
			}
		}
		set_temp_variable = {
			SAM_mobile_groups_drone_targets_engaged = SAM_mobile_groups_drone_capacity
		}
		# Общий БК страны: ammunition_value_k хранится в тысячах единиц.
		# Один перехватываемый объект этим комплексом расходует 0.10k.
		OWNER = {
			set_temp_variable = {
				SAM_common_ammo_targets_affordable = ammunition_value_k
			}
			divide_temp_variable = {
				var = SAM_common_ammo_targets_affordable
				value = 0.1
			}
			# floor(ammunition_value_k / cost), чтобы не получать дробные цели.
			set_temp_variable = {
				SAM_common_ammo_targets_fraction = SAM_common_ammo_targets_affordable
			}
			modulo_temp_variable = {
				var = SAM_common_ammo_targets_fraction
				value = 1
			}
			subtract_from_temp_variable = {
				var = SAM_common_ammo_targets_affordable
				value = SAM_common_ammo_targets_fraction
			}
			# Ограничиваем число целей номинальной capacity системы и остатком Raid.
			clamp_temp_variable = {
				var = SAM_common_ammo_targets_affordable
				min = 0
				max = PREV.SAM_mobile_groups_drone_targets_engaged
			}
			clamp_temp_variable = {
				var = SAM_common_ammo_targets_affordable
				min = 0
				max = PREV.SAM_targets_remaining
			}

				PREV.SAM_mobile_groups_drone_targets_engaged = SAM_common_ammo_targets_affordable
