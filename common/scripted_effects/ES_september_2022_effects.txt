ES_september_2022_update_captured_bridges = {
	if = {
		limit = { original_tag = UKR ES_is_september_2022_start = yes }
		if = {
			limit = {
				controls_state = 253
				NOT = { has_global_flag = ES_september_2022_kherson_bridges_destroyed }
			}
			set_variable = { global.dnipro_obj_current_hp^22 = 0 }
			set_variable = { global.dnipro_obj_current_hp^23 = 0 }
			set_variable = { global.dnipro_obj_is_destroyed^22 = 1 }
			set_variable = { global.dnipro_obj_is_destroyed^23 = 1 }
			set_global_flag = SADOVOE_STRAIT_BLOCKED
			set_global_flag = ANTONOVKA_STRAIT_BLOCKED
			set_global_flag = ES_september_2022_kherson_bridges_destroyed
			log = "[GetDateText]: September scenario: Ukraine captured state 253; both Kherson bridges destroyed"
		}
		if = {
			limit = {
				controls_state = 231
				NOT = { has_global_flag = ES_september_2022_kakhovka_bridge_destroyed }
			}
			set_variable = { global.dnipro_obj_current_hp^21 = 0 }
			set_variable = { global.dnipro_obj_is_destroyed^21 = 1 }
			set_global_flag = KAKHOVKA_STRAIT_BLOCKED
			set_global_flag = ES_september_2022_kakhovka_bridge_destroyed
			log = "[GetDateText]: September scenario: Ukraine captured state 231; Kakhovka crossing destroyed"
		}
	}
}
