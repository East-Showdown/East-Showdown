RUS_FSPOZ_upgrade = {
	if = {
		limit = {
			NOT = { has_idea = RUS_FSPOZ_5 }
		}
		if = {
			limit = { has_idea = RUS_FSPOZ_4 }
			swap_ideas = {
				remove_idea = RUS_FSPOZ_4
				add_idea = RUS_FSPOZ_5
			}
		}
		else_if = {
			limit = { has_idea = RUS_FSPOZ_3 }
			swap_ideas = {
				remove_idea = RUS_FSPOZ_3
				add_idea = RUS_FSPOZ_4
			}
		}
		else_if = {
			limit = { has_idea = RUS_FSPOZ_2 }
			swap_ideas = {
				remove_idea = RUS_FSPOZ_2
				add_idea = RUS_FSPOZ_3
			}
		}
		else_if = {
			limit = { has_idea = RUS_FSPOZ_1 }
			swap_ideas = {
				remove_idea = RUS_FSPOZ_1
				add_idea = RUS_FSPOZ_2
			}
		}
		else = {
			add_ideas = RUS_FSPOZ_1
		}
	}
}
