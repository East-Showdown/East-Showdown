UKR_init_preformed_legion_decisions = {
	if = {
		limit = { has_start_date > 2022.04.14 }
		# 17802501: Kalinouski regiment. Its decision also grants 5 army XP.
		if = {
			limit = { NOT = { has_country_flag = UKR_regiment_Kastus_Kalinouski_completed } }
			army_experience = 5
			set_country_flag = UKR_regiment_Kastus_Kalinouski_completed
		}
		# 17805701, 15800601, 15800901: the first three international legions.
		set_country_flag = UKR_form_1st_international_defense_legion_ukraine_completed
		set_country_flag = UKR_form_2nd_international_defense_legion_ukraine_completed
		set_country_flag = UKR_form_3rd_international_defense_legion_ukraine_completed
		# 17804901-17804904: Freedom of Russia Legion.
		set_country_flag = UKR_form_LSR_completed
	}
	if = {
		limit = { has_start_date > 2022.09.05 }
		# 16000201, 18809301: Russian Volunteer Corps.
		set_country_flag = UKR_form_RDK_completed
	}
}
