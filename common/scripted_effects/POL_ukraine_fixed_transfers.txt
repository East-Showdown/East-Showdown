# Exact named-variant transfers from the Polish reserve to Ukraine.
# Keep producer and variant_name together: Polish designs are POL-owned;
# the M1A2 SEP v3 supplied to Poland is an NTO design.
# Debit first and measure the actual change. A short debit is rolled back;
# only a complete shipment is credited to Ukraine and the aid ledger.

POL_transfer_fixed_BMP_to_ukraine = {
	if = {
		limit = { country_exists = UKR }
		set_temp_variable = { POL_fixed_transfer_removed = num_equipment@ifv_sov_bmp1_equipment }
		add_equipment_to_stockpile = {
			type = ifv_sov_bmp1_equipment
			variant_name = "BWP-1"
			producer = POL
			amount = -50
		}
		subtract_from_temp_variable = { POL_fixed_transfer_removed = num_equipment@ifv_sov_bmp1_equipment }
		if = {
			limit = { check_variable = { POL_fixed_transfer_removed = 50 } }
			UKR = {
				add_equipment_to_stockpile = {
					type = ifv_sov_bmp1_equipment
					variant_name = "BWP-1"
					producer = POL
					amount = 50
				}
			}
			add_to_variable = { _POL_weapons_list_ifv_equipment_var = 50 }
		}
		else = {
			if = {
				limit = { check_variable = { POL_fixed_transfer_removed > 0 } }
				add_equipment_to_stockpile = {
					type = ifv_sov_bmp1_equipment
					variant_name = "BWP-1"
					producer = POL
					amount = POL_fixed_transfer_removed
				}
			}
			add_political_power = 10
			log = "[GetDateText]: POL fixed shipment BWP-1 failed: insufficient exact variant; political power refunded"
		}
	}
	else = { add_political_power = 10 }
}

# D-30 has its own equipment type and needs no named designer variant.
POL_transfer_fixed_artillery_to_ukraine = {
	if = {
		limit = {
			country_exists = UKR
			has_equipment = { sov_artillery_d30_equipment > 49 }
		}
		send_equipment = {
			equipment = sov_artillery_d30_equipment
			amount = 50
			target = UKR
		}
		add_to_variable = { _POL_weapons_list_artillery_equipment_var = 50 }
	}
	else = {
		add_political_power = 10
		log = "[GetDateText]: POL fixed shipment D-30 failed: insufficient stock or recipient missing; political power refunded"
	}
}
