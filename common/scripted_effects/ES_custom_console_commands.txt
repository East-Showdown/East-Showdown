# round_test - show how round_temp_variable rounds (the substation formulas expect nearest)
d_round_test = {
	set_temp_variable = { r1 = 0.4 }
	round_temp_variable = r1
	set_temp_variable = { r2 = 0.5 }
	round_temp_variable = r2
	set_temp_variable = { r3 = 0.6 }
	round_temp_variable = r3
	set_temp_variable = { r4 = 1.5 }
	round_temp_variable = r4
	# floor pattern: x - 0.499, ceil pattern: x + 0.499
	set_temp_variable = { r5 = 0.501 }
	round_temp_variable = r5
	set_temp_variable = { r6 = 2.499 }
	round_temp_variable = r6
	set_temp_variable = { r7 = 2.5 }
	round_temp_variable = r7
	log = "SUBSTATION TEST: round 0.4=[?r1|.0] 0.5=[?r2|.0] 0.6=[?r3|.0] 1.5=[?r4|.0] | floor(1.0)=[?r5|.0] (want 1), ceil(2.0)=[?r6|.0] (want 2), ceil(2.001)=[?r7|.0] (want 3, 2 only matters for demands 0.001 above a whole substation)"

	if = {
		limit = {
			check_variable = { r1 = 0 }
			check_variable = { r3 = 1 }
			check_variable = { r5 = 1 }
			check_variable = { r6 = 2 }
		}
		log = "SUBSTATION TEST: PASS - round_temp_variable rounds to nearest, the substation formulas are correct"
	}
	else_if = {
		limit = { check_variable = { r3 = 0 } }
		log = "SUBSTATION TEST: FAIL - round_temp_variable rounds DOWN, the substation formulas must be changed"
	}
	else = {
		log = "SUBSTATION TEST: FAIL - unexpected rounding, see the values above"
	}
}

}

# depot_mod_clear <state> - remove the test modifiers again
d_depot_mod_clear = {
	if = { limit = { check_variable = { args^num < 1 } }
		log = "DEPOT TEST: usage: depot_mod_clear <state id>"
	}
	else = {
		set_temp_variable = { ds = args^0 }
		var:ds = {
			remove_dynamic_modifier = { modifier = ammo_depot_strike_3 scope = RUS }
			remove_dynamic_modifier = { modifier = ammo_depot_strike_3 scope = DPR }
			remove_dynamic_modifier = { modifier = ammo_depot_strike_3 scope = LPR }
			remove_dynamic_modifier = { modifier = ammo_depot_strike_3 scope = WGN }
			es_depot_test_print = yes
		}
	}
}

# depot_front_info - count the states flagged as front line for field depot strikes (ammo_depot_front)
d_depot_front_info = {
	set_temp_variable = { dft_all = 0 }
	set_temp_variable = { dft_rus = 0 }
	set_temp_variable = { dft_ukr = 0 }
	every_state = {
		limit = { has_state_flag = ammo_depot_front }
		add_to_temp_variable = { dft_all = 1 }
		if = { limit = { controller = { tag = RUS } } add_to_temp_variable = { dft_rus = 1 } }
		if = { limit = { controller = { tag = UKR } } add_to_temp_variable = { dft_ukr = 1 } }
	}
	log = "DEPOT TEST: front-line states flagged: [?dft_all|0] (controlled by RUS [?dft_rus|0], UKR [?dft_ukr|0])"
}
