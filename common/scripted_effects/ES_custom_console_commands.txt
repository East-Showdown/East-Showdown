# =====================================================================
# Bridge / dam test commands (see console_commands.md)
# <pos> = object position 0-28, the same number used in raids and
# decisions. Output is written to the console and to game.log.
# Run bridge_list first to see every position.
# =====================================================================

# --- helpers (not commands) ---
es_bridge_console_check_pos = {
	set_temp_variable = { b_ok = 0 }
	if = {
		limit = {
			check_variable = { bpos > -1 }
			check_variable = { bpos < global.dnipro_obj_max_hp^num }
		}
		set_temp_variable = { b_ok = 1 }
	}
	else = {
		log = "BRIDGE TEST: position [?bpos|.0] does not exist, use 0 - [?global.dnipro_obj_max_hp^num|.0] minus one"
	}
}

es_bridge_console_print = {
	set_temp_variable = { b_tok = global.dnipro_objects^bpos }
	set_temp_variable = { b_flag = global.dnipro_obj_strait^bpos }
	set_temp_variable = { b_hp = global.dnipro_obj_current_hp^bpos }
	set_temp_variable = { b_max = global.dnipro_obj_max_hp^bpos }
	set_temp_variable = { b_crit = global.dnipro_obj_critical_hp^bpos }
	set_temp_variable = { b_start = global.dnipro_obj_start_loss_hp^bpos }
	set_temp_variable = { b_loss = global.dnipro_obj_damage^bpos }
	set_temp_variable = { b_dead = global.dnipro_obj_is_destroyed^bpos }
	set_temp_variable = { b_owner = global.bridges_province_controllers^bpos }
	set_temp_variable = { b_blocked = 0 }
	meta_effect = {
		F = "[?b_flag.GetTokenKey]"
		text = {
			if = { limit = { has_global_flag = [F] } set_temp_variable = { b_blocked = 1 } }
		}
	}
	log = "BRIDGE TEST: [?bpos|.0] [?b_tok.GetTokenKey] - HP [?b_hp|.1] / [?b_max|.0], blocks below [?b_crit|.1], blocked [?b_blocked|.0], destroyed [?b_dead|.0], erodes below [?b_start|.0] by [?b_loss|.1] per week, owner [?b_owner|.0] (1 UKR, 2 RUS), flag [?b_flag.GetTokenKey]"
}

# Apply the instant part of a weekly tick: clamp HP, blocking flags, destruction news.
# No erosion and no repair, so the HP you set stays exact.
es_bridge_console_apply = {
	for_each_loop = {
		array = global.dnipro_obj_max_hp
		index = i
		clamp_variable = { var = global.dnipro_obj_current_hp^i min = 0 max = global.dnipro_obj_max_hp^i }
	}
	BOR = {
		autoblocking_all_dnipro_straits_objects = yes
		execute_destruction_scenario_dnipro_objects = yes
	}
}

# --- commands ---

# bridge_list - print every bridge/dam
d_bridge_list = {
	log = "BRIDGE TEST: ---- [?global.dnipro_obj_max_hp^num|.0] objects ----"
	for_each_loop = {
		array = global.dnipro_obj_max_hp
		index = list_i
		set_temp_variable = { bpos = list_i }
		es_bridge_console_print = yes
	}
}

# bridge_info <pos> - print one bridge/dam
d_bridge_info = {
	set_temp_variable = { bpos = args^0 }
	es_bridge_console_check_pos = yes
	if = { limit = { check_variable = { b_ok = 1 } }
		es_bridge_console_print = yes
	}
}

# bridge_hp <pos> <hp> - set HP and apply blocking/destruction immediately
d_bridge_hp = {
	set_temp_variable = { bpos = args^0 }
	set_temp_variable = { b_new = args^1 }
	es_bridge_console_check_pos = yes
	if = { limit = { check_variable = { b_ok = 1 } }
		set_variable = { global.dnipro_obj_current_hp^bpos = b_new }
		es_bridge_console_apply = yes
		es_bridge_console_print = yes
	}
}

# bridge_damage <pos> <amount> - subtract HP exactly like a raid does (no update until the weekly tick)
d_bridge_damage = {
	set_temp_variable = { bpos = args^0 }
	set_temp_variable = { b_amount = args^1 }
	es_bridge_console_check_pos = yes
	if = { limit = { check_variable = { b_ok = 1 } }
		subtract_from_variable = { global.dnipro_obj_current_hp^bpos = b_amount }
		es_bridge_console_print = yes
	}
}

# bridge_week [n] - run the full weekly update n times (default 1): erosion, ownership, repair, blocking, news
d_bridge_week = {
	set_temp_variable = { b_weeks = args^0 }
	if = { limit = { check_variable = { b_weeks < 1 } } set_temp_variable = { b_weeks = 1 } }
	for_loop_effect = {
		end = b_weeks
		value = week_i
		BOR = { update_all_dnipro_objects = yes }
	}
	log = "BRIDGE TEST: ran [?b_weeks|.0] weekly update(s)"
}

# bridge_reset <pos> - full HP, clear the destroyed mark (news can fire again), reapply blocking
d_bridge_reset = {
	set_temp_variable = { bpos = args^0 }
	es_bridge_console_check_pos = yes
	if = { limit = { check_variable = { b_ok = 1 } }
		set_variable = { global.dnipro_obj_current_hp^bpos = global.dnipro_obj_max_hp^bpos }
		set_variable = { global.dnipro_obj_is_destroyed^bpos = 0 }
		es_bridge_console_apply = yes
		es_bridge_console_print = yes
	}
}

	FROM = { set_variable = { blowing_bridges_decisions_is_open = 1 } }

}
