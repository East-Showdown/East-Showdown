UKR_init_internal_refugees = {
	if = {
		limit = {
			has_war_with = RUS
			NOT = { has_country_flag = UKR_internal_refugees_initialized }
		}
		if = {
			limit = {
				NOT = { has_dynamic_modifier = { modifier = UKR_internal_refugees } }
			}
			set_variable = { UKR_internal_refugees_stability_weekly = -0.005 }
			set_variable = { UKR_internal_refugees_political_power_cost = 1 }
			set_variable = { UKR_internal_refugees_stability_factor = -0.075 }
			set_variable = { UKR_internal_refugees_war_support_factor = -0.05 }
			set_variable = { UKR_internal_refugees_consumer_goods_factor = 0.015 }
			set_variable = { UKR_internal_refugees_monthly_population = -0.05 }
			set_variable = { UKR_internal_refugees_resistance_growth_on_our_occupied_states = -0.1 }

			if = {
				limit = { has_completed_focus = UKR_help_internal_refugees }
				add_to_variable = { UKR_internal_refugees_stability_weekly = 0.005 }
				add_to_variable = { UKR_internal_refugees_stability_factor = 0.05 }
				add_to_variable = { UKR_internal_refugees_consumer_goods_factor = 0.005 }
				add_to_variable = { UKR_internal_refugees_monthly_population = 0.025 }
			}
			if = {
				limit = { has_completed_focus = UKR_prykhystok }
				add_to_variable = { UKR_internal_refugees_stability_factor = 0.015 }
				add_to_variable = { UKR_internal_refugees_political_power_cost = -0.5 }
				add_to_variable = { UKR_internal_refugees_monthly_population = 0.01 }
			}

			add_dynamic_modifier = { modifier = UKR_internal_refugees }
			force_update_dynamic_modifier = yes
		}
		set_country_flag = UKR_internal_refugees_initialized
	}
}
