# has_start_date accepts strict comparisons, not equality. Keep both bounds
# in one trigger so NOT also negates the complete bookmark-date check.
ES_is_january_2022_start = {
	NOT = { has_start_date < 2022.01.01 }
	has_start_date < 2022.01.02
}

ES_is_april_2022_start = {
	NOT = { has_start_date < 2022.04.15 }
	has_start_date < 2022.04.16
}

ES_is_september_2022_start = {
	NOT = { has_start_date < 2022.09.06 }
	has_start_date < 2022.09.07
}
