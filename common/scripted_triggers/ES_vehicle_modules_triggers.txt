# Which national vehicle-module tree (vehicle_modules_folder) a country sees.
# Mirrors the per-origin branches of the infantry tab: every country sees exactly one tree.

is_vmod_UKR_tree = {
	OR = {
		is_UKR_origin = yes
		original_tag = CAU
	}
}

is_vmod_NTO_tree = {
	OR = {
		is_NTO_origin = yes
		is_POL_origin = yes
	}
}

is_vmod_RUS_tree = {
	NOT = {
		is_vmod_UKR_tree = yes
		is_vmod_NTO_tree = yes
	}
}
