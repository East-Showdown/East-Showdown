ES_auto_agency_can_start = {
	check_variable = { ES_auto_agency_choice > 1 }
	NOT = { has_active_mission = ES_auto_agency_upgrade_in_progress }
	# Выставляется каждым ванильным апгрейдом в modifiers_during_progress —
	# блокирует старт, пока идёт «родной» апгрейд, запущенный вне нашей системы.
	# [PROBE 298] временно отключено для проверки точки отказа:
	# check_variable = { modifier@ES_auto_agency_in_progress_boolean = 0 }
	check_variable = { ES_auto_agency_selected_upgrade = -1 }
	has_intelligence_agency = yes
	# Работаем, пока что-то не докачано, либо когда у игрока есть явная очередь.
	# Без этого вручную заказанные доп. уровни зависнут навсегда после первой прокачки.
	OR = {
		ES_auto_agency_can_upgrade = yes
		AND = {
			has_variable = ES_auto_agency_queue^num
			check_variable = { ES_auto_agency_queue^num > 0 }
		}
	}
}

# и защищают подстановку в meta_effect от чтения слотов вне допустимого диапазона.

	check_variable = { v > -1 }
