# num_battalions_in_states принимает только числа в тексте: стейт и порог подставляет meta_trigger.
# Переменная со стейтом хранит не номер: номер даёт GetID, формат [?var|0] выводит мусор

# В триггерах рейда; требует temp target_state, scope — страна-жертва
ammo_depot_state_has_consumers = {
	meta_trigger = {
		text = {
			num_battalions_in_states = {
				count > 0
				states = { [STATE] }
				types = { towed_art artillery sp_art support_spa sp_heavy_art support_heavy_spa MRLS support_mlrs TOS support_tos }
			}
		}
		STATE = "[?target_state.GetID]"
	}
}
