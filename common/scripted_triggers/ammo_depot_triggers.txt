# num_battalions_in_states принимает только числа в тексте: стейт и порог подставляет meta_trigger.
# Переменная со стейтом хранит не номер: номер даёт GetID, формат [?var|0] выводит мусор

# В триггерах рейда; требует temp target_state, scope — страна-жертва
ammo_depot_state_has_consumers = {
	meta_trigger = {
		text = {
			num_battalions_in_states = {
				count > 0
				states = { [STATE] }
				types = { towed_art artillery sp_art support_spa sp_heavy_art support_heavy_spa MRLS support_mlrs TOS support_tos }
			}
		}
		STATE = "[?target_state.GetID]"
	}
}

# Требует temp ammo_depot_state, ammo_depot_type (1-4), ammo_depot_probe; scope — страна-жертва.
# 2026-11-10: 4 группы с одинаковым расходом вместо 10 типов (каждая проверка перебирает всю армию страны)
# 1 — towed_art, artillery, sp_art, support_spa; 2 — sp_heavy_art, support_heavy_spa; 3 — MRLS, support_mlrs; 4 — TOS, support_tos
ammo_depot_more_than_probe = {
	OR = {
		AND = {
			check_variable = { ammo_depot_type = 1 }
			meta_trigger = {
				text = { num_battalions_in_states = { count > [COUNT] states = { [STATE] } types = { towed_art artillery sp_art support_spa } } }
				COUNT = "[?ammo_depot_probe|0]"
				STATE = "[?ammo_depot_state.GetID]"
			}
		}
		AND = {
			check_variable = { ammo_depot_type = 2 }
			meta_trigger = {
				text = { num_battalions_in_states = { count > [COUNT] states = { [STATE] } types = { sp_heavy_art support_heavy_spa } } }
				COUNT = "[?ammo_depot_probe|0]"
				STATE = "[?ammo_depot_state.GetID]"
			}
		}
		AND = {
			check_variable = { ammo_depot_type = 3 }
			meta_trigger = {
				text = { num_battalions_in_states = { count > [COUNT] states = { [STATE] } types = { MRLS support_mlrs } } }
				COUNT = "[?ammo_depot_probe|0]"
				STATE = "[?ammo_depot_state.GetID]"
			}
		}
		AND = {
			check_variable = { ammo_depot_type = 4 }
			meta_trigger = {
				text = { num_battalions_in_states = { count > [COUNT] states = { [STATE] } types = { TOS support_tos } } }
				COUNT = "[?ammo_depot_probe|0]"
				STATE = "[?ammo_depot_state.GetID]"
			}
		}
	}
}

# Стороны войны для ударов по полевым складам: решения и защита есть только у RUS и UKR,
# остальные участники войны пользуются защитой своего лидера (2026-11-10). Scope — страна
ammo_depot_rus_side = {
	OR = {
		tag = RUS
		has_war_with = UKR
	}
}
ammo_depot_ukr_side = {
	NOT = { ammo_depot_rus_side = yes }
	OR = {
		tag = UKR
		has_war_with = RUS
	}
}

# В триггерах рейда; scope — атакующий, требует temp target_state.
# Сначала дешёвая проверка контролёра (FROM), полный перебор стран — только если у него нет артиллерии в области
ammo_depot_enemy_consumers_in_target = {
	OR = {
		FROM = { ammo_depot_state_has_consumers = yes }
		any_country = {
			has_war_with = ROOT
			divisions_in_state = { state = var:target_state size > 0 }
			ammo_depot_state_has_consumers = yes
		}
	}
}
