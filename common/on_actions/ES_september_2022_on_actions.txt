on_actions = {
	on_state_control_changed = {
		effect = {
			if = {
				limit = { original_tag = UKR has_start_date = 2022.09.06 }
				ES_september_2022_update_captured_bridges = yes
			}
		}
	}
	on_daily = {
		effect = {
			if = {
				limit = {
					has_start_date = 2022.09.06
					country_exists = UKR
					NOT = { date < 2023.06.07 }
					date < 2023.06.08
					NOT = { has_global_flag = ES_september_2022_kakhovka_news_fired }
				}
				set_global_flag = ES_september_2022_kakhovka_news_fired
				UKR = { news_event = { id = es_september.1 } }
			}
		}
	}
}
