on_actions = {
	on_startup = {
		effect = {
			every_country = {
				init_wounded_system = yes
			}
		}
	}

	on_weekly = {
		effect = {
			calculate_healed_wounded = yes
			calculate_country_casualties = yes
			calculate_returned_prisoners_of_war = yes
		}
	}

	on_weekly_RUS = {
		effect = {
			if = {
				limit = { has_war = yes }
				calculate_prisoners_of_war_captured_by_UKR = yes
			}
		}
	}

	on_weekly_UKR = {
		effect = {
			if = {
				limit = { has_war = yes }
				calculate_prisoners_of_war_captured_by_RUS = yes
			}

			if = {
				limit = {
					check_variable = { var = casualties_k value = 50 compare = greater_than_or_equals }
					NOT = { has_country_flag = UKR_szch_started }
				}
				set_country_flag = UKR_szch_started
				add_ideas = UKR_szch_50
				country_event = { id = ukr_szch.1 days = 0 }
			}
			if = {
				limit = { has_country_flag = UKR_szch_started }
				UKR_update_szch_spirit = yes
			}
		}
	}
}
