on_actions = {
	on_startup = {
		effect = {
			UKR = { UKR_init_internal_refugees = yes }
		}
	}
	on_declare_war = {
		effect = {
			if = {
				limit = {
					OR = {
						AND = { tag = RUS FROM = { tag = UKR } }
						AND = { tag = UKR FROM = { tag = RUS } }
					}
				}
				UKR = { UKR_init_internal_refugees = yes }
			}
		}
	}
	on_war = {
		effect = {
			if = {
				limit = { tag = UKR }
				UKR_init_internal_refugees = yes
			}
		}
	}
	on_daily_UKR = {
		effect = {
			UKR_init_internal_refugees = yes
		}
	}
}
