on_actions = {
    on_startup = {
        effect = {
            every_country = {
				if = {
                    limit = {
                        tag = RUS
                        is_ai = yes
                    }
					add_political_power = 300
                }
            }
			RUS = {

				set_variable = {
					RUS_rosconservation_zapas = 10000
				}
				set_variable = {
					RUS_ammunition_warehouse_count_restore = 0
				}
			}
        }
    }

	on_monthly_RUS = {
		effect = {
			if = {
				limit = {
					has_war_with = UKR
					NOT = { has_idea = RUS_centralised_information_control }
					UKR = { surrender_progress < 0.4 }
					OR = {
						NOT = { has_variable = RUS_political_crisis_risk }
						check_variable = { RUS_political_crisis_risk < 100 }
					}
				}
				add_to_variable = { RUS_political_crisis_risk = 3.5 }
				RUS_recalculate_crisis_pressure = yes
			}
		}
	}
}
