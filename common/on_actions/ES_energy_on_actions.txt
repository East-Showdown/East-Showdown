on_actions = {
	on_startup = {
		effect = {
			BOR = {
				init_global_energy_data = yes
				init_all_hydro_power_plants = yes
			}

			# Create the blackout status read by more_mapmodes_blackout_map_mode on every state,
			# so later weekly updates only change the value
			every_state = {
				set_variable = { energy_blackout_status = 0 }
			}

			if = {
				limit = {
					has_game_rule = {
						rule = ES_ENERGY_SYSTEM_STATUS
						option = ES_ENERGY_SYSTEM_ENABLED
					}
				}

				every_country = {
					init_country_energy_data = yes
					add_dynamic_modifier = { modifier = energy_country_status }
				}
			}
		}
	}

	on_weekly = {
		effect = {
			if = {
				limit = {
					has_game_rule = {
						rule = ES_ENERGY_SYSTEM_STATUS
						option = ES_ENERGY_SYSTEM_ENABLED
					}
				}
				
				calculate_energy_in_country = yes
			}
		}
	}
}
