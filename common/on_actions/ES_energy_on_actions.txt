on_actions = {
	on_startup = {
		effect = {
			BOR = {
				init_global_energy_data = yes
				init_all_hydro_power_plants = yes
			}

			# Create the blackout status read by more_mapmodes_blackout_map_mode on every state,
			# so later weekly updates only change the value
			every_state = {
				set_variable = { energy_blackout_status = 0 }
			}

			if = {
				limit = {
					has_game_rule = {
						rule = ES_ENERGY_SYSTEM_STATUS
						option = ES_ENERGY_SYSTEM_ENABLED
					}
				}

				every_country = {
					init_country_energy_data = yes
					add_dynamic_modifier = { modifier = energy_country_status }
				}
			}
		}
	}

	on_weekly = {
		effect = {
			if = {
				limit = {
					has_game_rule = {
						rule = ES_ENERGY_SYSTEM_STATUS
						option = ES_ENERGY_SYSTEM_ENABLED
					}
				}
				
				calculate_energy_in_country = yes

				# Check the freshly calculated production penalty only for Ukraine.
				if = {
					limit = {
						tag = UKR
						NOT = { has_country_flag = UKR_EnergyNightmare }
					}
					if = {
						limit = {
							check_variable = {
								var = energy_blackout_industrial_capacity_factory_factor
								value = -0.20
								compare = less_than_or_equals
							}
						}
						set_country_flag = UKR_EnergyNightmare
						mark_focus_tree_layout_dirty = yes
						news_event = { id = news.194 }
					}
				}
			}
		}
	}
}
