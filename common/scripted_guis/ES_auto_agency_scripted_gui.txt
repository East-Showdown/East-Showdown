scripted_gui = {
	ES_auto_agency_queue_scripted_gui = {
		context_type = player_context
		window_name = "ES_auto_agency_queue_gui"

		dirty = global.ES_auto_agency_dirty

		visible = {
			has_country_flag = ES_auto_agency_queue_gui_open
		}

		triggers = {
			decrease_order_click_enabled = {
				set_temp_variable = { last_in_array = ES_auto_agency_queue^num }
				subtract_from_temp_variable = { last_in_array = 1 }
				check_variable = { i < last_in_array }
			}
			decrease_order_shift_click_enabled = {
				set_temp_variable = { last_in_array = ES_auto_agency_queue^num }
				subtract_from_temp_variable = { last_in_array = 1 }
				check_variable = { i < last_in_array }
			}
			increase_order_click_enabled = {
				check_variable = { i > 0 }
			}
			increase_order_shift_click_enabled = {
				check_variable = { i > 0 }
			}

			agency_upgrade_icon_button_click_enabled = {
				custom_trigger_tooltip = {
					tooltip = ES_auto_agency_queue_not_full_tt
					NOT = {
						check_variable = {
							var = ES_auto_agency_upgrades_completed_and_queued^i
							value = global.ES_auto_agency_max_upgrades^i
							compare = greater_than_or_equals
						}
					}
				}
				custom_trigger_tooltip = {
					tooltip = ES_auto_agency_prereqs_tt
					OR = {
						AND = {
							NOT = { check_variable = { i = 14 } }
							NOT = { check_variable = { i = 15 } }
							NOT = { check_variable = { i = 16 } }
						}
						AND = {
							OR = {
								check_variable = { i = 14 }
								check_variable = { i = 15 }
							}
							has_done_agency_upgrade = upgrade_form_department
						}
						AND = {
							check_variable = { i = 16 }
							has_done_agency_upgrade = upgrade_crypto_strength
						}
					}
				}
			}
		}

		effects = {
			close_button_click = {
				clr_country_flag = ES_auto_agency_queue_gui_open
				ES_auto_agency_mark_dirty = yes
			}

			expand_agency_dropdown_button_click = {
				ES_auto_agency_toggle_dropdown = yes
				ES_auto_agency_mark_dirty = yes
			}
			expand_agency_dropdown_button_arrow_click = {
				ES_auto_agency_toggle_dropdown = yes
				ES_auto_agency_mark_dirty = yes
			}

			decrease_order_click = {
				set_temp_variable = { below = i }
				add_to_temp_variable = { below = 1 }
				set_temp_variable = { below_value = ES_auto_agency_queue^below }

				set_temp_variable = { hold = v }

				set_variable = { ES_auto_agency_queue^i = below_value }
				set_variable = { ES_auto_agency_queue^below = hold }
				ES_auto_agency_mark_dirty = yes
			}

			decrease_order_shift_click = {
				set_temp_variable = { hold_value = v }
				set_temp_variable = { shrink_idx = i }
				ES_auto_agency_shrink_queue_at_index = yes
				add_to_array = { ES_auto_agency_queue = hold_value }
				ES_auto_agency_mark_dirty = yes
			}

			increase_order_click = {
				set_temp_variable = { above = i }
				subtract_from_temp_variable = { above = 1 }
				set_temp_variable = { above_value = ES_auto_agency_queue^above }

				set_temp_variable = { hold = v }

				set_variable = { ES_auto_agency_queue^i = above_value }
				set_variable = { ES_auto_agency_queue^above = hold }
				ES_auto_agency_mark_dirty = yes
			}

			increase_order_shift_click = {
				set_temp_variable = { hold_value = v }

				for_loop_effect = {
					start = i
					end = 0
					value = v2
					compare = greater_than
					add = -1

					set_temp_variable = { above = v2 }
					subtract_from_temp_variable = { above = 1 }
					set_temp_variable = { above_value = ES_auto_agency_queue^above }

					set_variable = { ES_auto_agency_queue^v2 = above_value }
				}

				set_variable = { ES_auto_agency_queue^0 = hold_value }
				ES_auto_agency_mark_dirty = yes
			}

			remove_from_queue_click = {
				set_temp_variable = { val = ES_auto_agency_queue^i }
				subtract_from_variable = { ES_auto_agency_upgrades_completed_and_queued^val = 1 }
				set_temp_variable = { shrink_idx = i }
				ES_auto_agency_shrink_queue_at_index = yes
				country_event = { id = ES_auto_agency.1 days = 1 }
				ES_auto_agency_mark_dirty = yes
			}

			agency_upgrade_icon_button_click = {
				add_to_array = { ES_auto_agency_queue = i }
				add_to_variable = { ES_auto_agency_upgrades_completed_and_queued^i = 1 }
				ES_auto_agency_verify_queue = yes
				# Игрок снова наполнил очередь — снимаем ожидание после Clear.
				set_variable = { ES_auto_agency_queue_cleared = 0 }
				# === TEMP-DEBUG ES_AA_DBG: факт добавления в очередь (удалить после диагностики) ===
				log = "[ES_AA_DBG] QUEUE_ADD(click) | queue_num=[?ES_auto_agency_queue^num] cleared=[?ES_auto_agency_queue_cleared]"

				if = {
					limit = {
						OR = {
							NOT = { check_variable = { ES_auto_agency_choice = 3 } }
							has_active_mission = ES_auto_agency_upgrade_in_progress
							check_variable = { ES_auto_agency_queue^num > 1 }
						}
					}
					ES_auto_agency_update_state = yes
				}
				else = {
					# Queue-Only + пустая очередь: откладываем, чтобы элемент
					# отрисовался до того, как find_first_valid его съест.
					country_event = { id = ES_auto_agency.1 days = 1 }
				}
				ES_auto_agency_mark_dirty = yes
			}

			agency_upgrade_icon_button_shift_click = {
				set_temp_variable = { loop_duration = global.ES_auto_agency_max_upgrades^i }
				subtract_from_temp_variable = { loop_duration = ES_auto_agency_upgrades_completed_and_queued^i }
				set_temp_variable = { upgrade_index = i }

				if = {
					limit = { check_variable = { loop_duration > 0 } }

					for_loop_effect = {
						end = loop_duration

						add_to_array = { ES_auto_agency_queue = upgrade_index }
						add_to_variable = { ES_auto_agency_upgrades_completed_and_queued^upgrade_index = 1 }
					}

					ES_auto_agency_verify_queue = yes
					# Игрок снова наполнил очередь — снимаем ожидание после Clear.
					set_variable = { ES_auto_agency_queue_cleared = 0 }
					# === TEMP-DEBUG ES_AA_DBG: факт добавления в очередь (удалить после диагностики) ===
					log = "[ES_AA_DBG] QUEUE_ADD(shift) | queue_num=[?ES_auto_agency_queue^num] cleared=[?ES_auto_agency_queue_cleared]"

					if = {
						limit = {
							OR = {
								NOT = { check_variable = { ES_auto_agency_choice = 3 } }
								has_active_mission = ES_auto_agency_upgrade_in_progress
								check_variable = { ES_auto_agency_queue^num > 1 }
							}
						}
						ES_auto_agency_update_state = yes
					}
					else = {
						country_event = { id = ES_auto_agency.1 days = 1 }
					}

					ES_auto_agency_mark_dirty = yes
				}
			}

			clear_queue_button_click = {
				for_each_loop = {
					array = ES_auto_agency_queue
					value = upgrade_idx
					index = local_index

					subtract_from_variable = { ES_auto_agency_upgrades_completed_and_queued^upgrade_idx = 1 }
				}

				clear_array = ES_auto_agency_queue
				# Очередь очищена вручную: ON не должен сам выбирать случайный
				# апгрейд, пока игрок снова не заполнит очередь.
				set_variable = { ES_auto_agency_queue_cleared = 1 }
				country_event = { id = ES_auto_agency.1 days = 1 }
				ES_auto_agency_mark_dirty = yes
			}
		}

		properties = {
			agency_queue_icon = {
				image = "[ES_auto_agency_queue_icon]"
			}
			agency_upgrade_icon_button = {
				image = "[ES_auto_agency_upgrade_icon]"
			}
			expand_agency_dropdown_button_arrow = {
				frame = ES_auto_agency_dropdown_visible
			}
		}

		dynamic_lists = {
			queue_list_gridbox = {
				array = ES_auto_agency_queue
				change_scope = no
				entry_container = "ES_auto_agency_upgrade_in_queue_item"
			}
			upgrades_list_gridbox = {
				array = global.ES_auto_agency_upgrades
				change_scope = no
				entry_container = "ES_auto_agency_upgrade_item"
			}
		}
	}

	ES_auto_agency_upgrade_selection_option_dropdown_gui = {
		context_type = player_context
		window_name = "ES_auto_agency_upgrade_selection_option_dropdown_container"
		parent_window_name = ES_auto_agency_queue_gui

		dirty = global.ES_auto_agency_dirty

		visible = {
			has_country_flag = ES_auto_agency_queue_gui_open
			check_variable = { ES_auto_agency_dropdown_visible = 1 }
		}

		effects = {
			auto_upgrades_off_button_click = {
				set_variable = { ES_auto_agency_choice = 1 }
				set_variable = { ES_auto_agency_dropdown_visible = 2 }
				ES_auto_agency_update_state = yes
			}
			auto_upgrades_on_button_click = {
				set_variable = { ES_auto_agency_choice = 2 }
				set_variable = { ES_auto_agency_dropdown_visible = 2 }
				# Отложено, чтобы find_first_valid не срезал очередь в тот же тик,
				# когда игрок переключил автоматизацию.
				country_event = { id = ES_auto_agency.1 days = 1 }
				ES_auto_agency_mark_dirty = yes
			}
			auto_upgrades_queue_only_button_click = {
				set_variable = { ES_auto_agency_choice = 3 }
				set_variable = { ES_auto_agency_dropdown_visible = 2 }
				country_event = { id = ES_auto_agency.1 days = 1 }
				ES_auto_agency_mark_dirty = yes
			}
		}
	}

	ES_auto_agency_upgrade_automation_scripted_ui = {
		context_type = player_context
		window_name = "ES_auto_agency_upgrade_automation_ui"
		parent_window_name = agency_branches

		dirty = global.ES_auto_agency_dirty

		visible = {
			has_intelligence_agency = yes
		}

		effects = {
			load_queue_tab_button_click = {
				# Самолечение глобалов: on_startup не срабатывает при загрузке
				# сейва, из-за чего массивы апгрейдов могли остаться пустыми и
				# сетка отрисовалась бы пустой. Создаём их при первом открытии.
				if = {
					limit = { NOT = { has_variable = global.ES_auto_agency_upgrades^num } }
					ES_auto_agency_init_globals = yes
				}
				else_if = {
					limit = { check_variable = { global.ES_auto_agency_upgrades^num < 17 } }
					ES_auto_agency_init_globals = yes
				}
				# Ленивая инициализация — пропуск on_startup экономит ~5000
				# переменных на AI-нациях ради player-only фичи.
				if = {
					limit = { NOT = { has_variable = ES_auto_agency_choice } }
					ES_auto_agency_init_country_state = yes
				}
				if = {
					limit = {
						has_country_flag = ES_auto_agency_queue_gui_open
					}
					clr_country_flag = ES_auto_agency_queue_gui_open
				}
				else = {
					set_country_flag = ES_auto_agency_queue_gui_open
				}
				ES_auto_agency_mark_dirty = yes
			}
		}
	}
}
