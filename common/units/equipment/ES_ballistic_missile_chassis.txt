		module_slots = {
			fixed_main_weapon_slot = {
				required = yes
				allowed_module_categories = {

				}
			}

			propulsion_slot = {
				required = yes
				allowed_module_categories = {
					rus_ballistic_engine

				}
			}

			guidance_slot = {
				required = yes
				allowed_module_categories = {
					rus_missile_guidance_Glonass

				}
			}

			body_slot = {
				required = no
				allowed_module_categories = {
					missile_body
				}
			}

			electronics_slot = {
				required = no
				allowed_module_categories = {
					rus_missile_electronic
				}
			}

			special_type_slot_1 = {
				required = no
				allowed_module_categories = {
					missile_special
				}
			}
			special_type_slot_2 = {
				required = no
				allowed_module_categories = {
					missile_special
				}
			}
			special_type_slot_3 = {
				required = no
				allowed_module_categories = {
					missile_special
				}
			}
		}

		default_modules = {
			fixed_main_weapon_slot = empty
			guidance_slot = empty
			propulsion_slot = empty
		}

		reliability = 0.85
		naval_strike_attack = 0.5
		naval_strike_targetting = 0.2
		lend_lease_cost = 2.0
		build_cost_ic = 40
		resources = {
			aluminium = 2
			tungsten = 2
		}
		fuel_consumption = 0
