# Вес роли управляет желаемым составом армии и набором, а не производством снаряжения.
ES_UKR_recruit_assault_battalions = {
	allowed = { original_tag = UKR }
	enable = {
		is_ai = yes
		has_template = "Shturmovoy batalion (BTR)"
		has_country_flag = UKR_assault_template_editing_unlocked
		has_tech = assault_infantry
	}
	abort_when_not_enabled = yes
	ai_strategy = {
		type = role_ratio
		id = ES_assault_recruitment
		value = 40
	}
}
