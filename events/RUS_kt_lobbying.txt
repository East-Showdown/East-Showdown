country_event = {
	id = RUS_kt_lobbying.1
	title = RUS_kt_lobbying.1.t
	desc = RUS_kt_lobbying.1.d
	is_triggered_only = yes
	trigger = { is_ai = no }
	fire_only_once = yes
	option = {
		name = RUS_kt_lobbying.1.a
		trigger = { NOT = { has_political_power < 40 } }
		ai_chance = { factor = 75 }
		if = {
			limit = { is_ai = no }
			add_political_power = -40
			custom_effect_tooltip = RUS_kt_lobbying.1.a_effects
			hidden_effect = {
				add_to_variable = { RUS_kt_influence^0 = 2 }
				add_to_variable = { RUS_kt_loyalty^0 = 1 }
				add_to_variable = { RUS_kt_mood^0 = 4 }
				add_to_variable = { RUS_kt_mood^5 = -2 }
				add_to_variable = { RUS_kt_mood^4 = -1 }
				add_timed_idea = { idea = RUS_kt_lobby_security_1_accept days = 90 }
				RUS_kt_initialize = yes
			}
		}
	}
	option = {
		name = RUS_kt_lobbying.1.b
		ai_chance = { factor = 25 }
		if = {
			limit = { is_ai = no }
			custom_effect_tooltip = RUS_kt_lobbying.1.b_effects
			hidden_effect = {
				add_to_variable = { RUS_kt_loyalty^0 = -2 }
				add_to_variable = { RUS_kt_mood^0 = -4 }
				if = {
					limit = { check_variable = { RUS_kt_influence^0 < 31 } }
					add_timed_idea = { idea = RUS_kt_lobby_security_1_refuse days = 60 }
				}
				else_if = {
					limit = { check_variable = { RUS_kt_influence^0 < 60 } }
					add_timed_idea = { idea = RUS_kt_lobby_security_1_refuse_x150 days = 60 }
				}
				else_if = {
					limit = { check_variable = { RUS_kt_influence^0 < 80 } }
					add_timed_idea = { idea = RUS_kt_lobby_security_1_refuse_x175 days = 60 }
				}
				else = {
					add_timed_idea = { idea = RUS_kt_lobby_security_1_refuse_x200 days = 60 }
				}
				RUS_kt_initialize = yes
			}
		}
	}
}

	title = RUS_kt_lobbying.21.t
	desc = RUS_kt_lobbying.21.d
	is_triggered_only = yes
	trigger = { is_ai = no }
	fire_only_once = yes
	option = {
		name = RUS_kt_lobbying.21.a
		trigger = { NOT = { has_political_power < 35 } }
		ai_chance = { factor = 75 }
		if = {
			limit = { is_ai = no }
			add_political_power = -35
			custom_effect_tooltip = RUS_kt_lobbying.21.a_effects
			hidden_effect = {
				add_to_variable = { RUS_kt_influence^5 = 2 }
				add_to_variable = { RUS_kt_loyalty^5 = 2 }
				add_to_variable = { RUS_kt_mood^5 = 4 }
				add_to_variable = { RUS_kt_mood^0 = -2 }
				add_to_variable = { RUS_kt_mood^2 = 1 }
				add_timed_idea = { idea = RUS_kt_lobby_oligarchs_1_accept days = 90 }
				RUS_kt_initialize = yes
			}
		}
	}
	option = {
		name = RUS_kt_lobbying.21.b
		ai_chance = { factor = 25 }
		if = {
			limit = { is_ai = no }
			custom_effect_tooltip = RUS_kt_lobbying.21.b_effects
			hidden_effect = {
				add_to_variable = { RUS_kt_loyalty^5 = -2 }
				add_to_variable = { RUS_kt_mood^5 = -5 }
				if = {
					limit = { check_variable = { RUS_kt_influence^5 < 31 } }
					add_timed_idea = { idea = RUS_kt_lobby_oligarchs_1_refuse days = 60 }
				}
				else_if = {
					limit = { check_variable = { RUS_kt_influence^5 < 60 } }
					add_timed_idea = { idea = RUS_kt_lobby_oligarchs_1_refuse_x150 days = 60 }
				}
				else_if = {
					limit = { check_variable = { RUS_kt_influence^5 < 80 } }
					add_timed_idea = { idea = RUS_kt_lobby_oligarchs_1_refuse_x175 days = 60 }
				}
				else = {
					add_timed_idea = { idea = RUS_kt_lobby_oligarchs_1_refuse_x200 days = 60 }
				}
				RUS_kt_initialize = yes
			}
		}
	}
}
